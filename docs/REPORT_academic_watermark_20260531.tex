% ── Abstract ──────────────────────────────────────────────────────────────
\begin{abstract}
\noindent
We present a rigorous empirical comparison of two watermark strategies for out-of-order stream processing --- \textit{strict watermarks} and \textit{heuristic watermarks with dead-letter queues (DLQ)} --- on a large-scale dataset of 2.96 million events derived from NYC Yellow Taxi trips.
Using time-compressed event/arrival timestamps ($\text{DIV}=60$), we characterize the \textbf{data completeness \% vs.\ wait time (ms)} trade-off through 15 $\delta$ configurations, 6 $p$-percentile adaptive configurations, and sensitivity analysis across 5 window sizes (1--60\,s).

We establish that the heuristic $+$ DLQ strategy is \textbf{strictly Pareto-dominant} for eventual completeness, achieving 100\% event capture with watermark latencies as low as 11.6\,s (median). Conversely, strict watermarks require $\delta \geq 120$\,s for near-total capture (99.97\%), permanently dropping 833 events beyond that threshold.

We derive the marginal cost function $C'(\delta) = \partial(\text{wait})/\partial(\text{completeness})$, identify exponential growth of cost beyond the p95 lateness threshold ($C' > 10^4$ ms/\% gain for $\delta > 60$\,s), and provide formal decision bounds for real-time vs.\ batch deployment scenarios.

\vspace{4pt}
\noindent\textbf{Statistical Methods:} Bootstrap CI ($n{=}1000$), MLE distribution fitting, GEV extreme-value analysis, KS goodness-of-fit.
\end{abstract}

\section{Introduction}
% ===========================================================================

\subsection{Motivation}

Distributed stream processing engines face a fundamental tension between \textbf{result completeness} and \textbf{output latency} when handling out-of-order events.
The \textit{watermark} --- an assertion about the progress of event time --- is the primary mechanism for bounding this trade-off.
In windowed aggregations, the watermark determines when a window is deemed ``closed'' and its results emitted.
Events arriving after the watermark are either dropped (permanent data loss) or rerouted to a dead-letter queue (delayed reconciliation).

Two families of watermark strategies have emerged:

\begin{enumerate}[leftmargin=*, itemsep=3pt]
\item \textbf{Strict Watermarks:}
  The watermark is a deterministic function of observed event times, parameterized by a \textit{safety margin} $\delta$ ($\texttt{DELTA\_BASE\_S}$).
  At any moment $t$, $W(t) = \max(\text{event\_time\_seen}) - \delta$.
  Every unit of $\delta$ contributes directly to end-to-end latency but potentially recovers more late events.
  Once a window is closed, late events are \textbf{permanently discarded}.

\item \textbf{Heuristic Watermarks:}
  The watermark adapts to the observed lateness distribution using a streaming quantile estimator (DDSketch).
  The safety margin $\Leff$ is the $p$-th percentile of recent lateness:
  $W_h(t) = \max(\text{event\_time\_seen}) - \Leff$.
  Events that arrive after the watermark are \textbf{routed to a Dead Letter Queue (DLQ)} for eventual reconciliation, enabling 100\% \textit{eventual} completeness.
\end{enumerate}

\subsection{Contributions}

This report provides:
\begin{enumerate}[leftmargin=*, itemsep=1pt]
\item A \textbf{formal mathematical framework} for comparing watermark strategies via the \textit{completeness-latency Pareto frontier}
\item \textbf{Empirical characterization} of $\Cstrict(\delta)$ with 15 data points, bootstrap confidence intervals, and distribution fitting
\item \textbf{Streaming simulation} of heuristic watermark behavior with DDSketch-like adaptive $\Leff$ tracking
\item \textbf{Sensitivity analysis} showing how window size $\omega$ modulates the completeness curve
\item \textbf{Marginal cost function} capturing the economic trade-off between latency and completeness
\item \textbf{Extreme-value analysis} of the lateness tail via block-maxima GEV fitting
\item \textbf{Information-theoretic characterization} of uncertainty reduction by different $\delta$ choices
\item \textbf{Prescriptive deployment recommendations} grounded in quantitative analysis
\end{enumerate}

\section{Formal Problem Statement}
% ===========================================================================

\subsection{Definitions}

Let $\mathcal{E} = \{e_1, e_2, \ldots, e_N\}$ be a stream of $N$ events. Each event $e_i$ carries:

\begin{itemize}[leftmargin=*, itemsep=1pt]
  \item \textbf{Event time} $\eventtime_i \in \mathbb{R}^+$: logical occurrence time
  \item \textbf{Arrival time} $\arrivaltime_i \in \mathbb{R}^+$: processing-time observation instant
  \item \textbf{Lateness} $\lateness_i = \arrivaltime_i - \eventtime_i \geq 0$: observation delay
  \item \textbf{Partition key} $k_i$: distributed sharding key
  \item \textbf{Window assignment}: $e_i$ belongs to tumbling window $[w_s, w_s + \omega)$ iff $w_s \leq \eventtime_i < w_s + \omega$
\end{itemize}

\vspace{4pt}
\noindent\textbf{Definition 1 (Strict Watermark).}
For safety margin $\delta \in \mathbb{R}^+$, the strict watermark at position $i$:
\begin{equation}
  W_{\text{strict}}(i, \delta) = \max_{j < i} \eventtime_j - \delta
  \label{eq:strict}
\end{equation}
Event $e_i$ is \textbf{on-time} iff:
\begin{equation}
  W_{\text{strict}}(i, \delta) = -\infty \;\lor\; \big\lfloor \eventtime_i / \omega \big\rfloor \cdot \omega + \omega > W_{\text{strict}}(i, \delta)
  \label{eq:strict-test}
\end{equation}

\vspace{4pt}
\noindent\textbf{Definition 2 (Heuristic Watermark with DLQ).}
Let $\hat{Q}_p$ be a streaming estimate of the $p$-th percentile of $\{\lateness_1, \ldots, \lateness_{i-1}\}$:
\begin{equation}
  W_{\text{heur}}(i, p) = \max_{j < i} \eventtime_j - \hat{Q}_p(\{\lateness_1, \ldots, \lateness_{i-1}\})
  \label{eq:heuristic}
\end{equation}
Events failing the test are routed to DLQ $\mathcal{D}$:
\begin{equation}
  \mathcal{D} = \big\{ e_i \mid \lfloor \eventtime_i / \omega \rfloor \cdot \omega + \omega \leq W_{\text{heur}}(i, p) \big\}
  \label{eq:dlq}
\end{equation}

\vspace{4pt}
\noindent\textbf{Definition 3 (Data Completeness).}
\begin{align}
  \Cimm &= \frac{|\{e_i \in \mathcal{E} : e_i \text{ passes watermark test}\}|}{N} \label{eq:cimm} \\[4pt]
  \Cev  &= \frac{|\{e_i \in \mathcal{E} : e_i \text{ passes watermark test} \lor e_i \in \mathcal{D}_{\text{reconciled}}\}|}{N} \label{eq:cev}
\end{align}
For strict watermark: $\Cev = \Cimm$ (no recovery).
For heuristic $+$ DLQ: $\Cev = 1.0$ (all DLQ events reconciled).

\vspace{4pt}
\noindent\textbf{Definition 4 (Wait Time).}
The \textit{wait time} $\tau$ is the additional latency beyond the window boundary:
\begin{itemize}[leftmargin=*, itemsep=1pt]
  \item \textbf{Strict}: $\tau_{\text{strict}} = \delta$ (safety margin added to window close delay)
  \item \textbf{Heuristic}: $\tau_{\text{heur}} = \Leff$ (adaptive effective lag)
\end{itemize}

\subsection{The Completeness-Latency Trade-off Function}

For strict watermarks, $\Cstrict: \mathbb{R}^+ \to [0, 1]$ maps $\delta$ to the fraction correctly classified:
\begin{equation}
  \Cstrict(\delta) = \frac{1}{N} \sum_{i=1}^{N} \indicator\!\left[ \big\lfloor \eventtime_i / \omega \big\rfloor \cdot \omega + \omega > \max_{j < i} \eventtime_j - \delta \right]
  \label{eq:cstrict}
\end{equation}

This admits a useful approximation via the lateness CDF:
\begin{equation}
  \Cstrict(\delta) \approx F_\ell(\delta) = \frac{1}{N} \sum_{i=1}^{N} \indicator[\lateness_i \leq \delta]
  \label{eq:approx}
\end{equation}
The approximation error arises from window-boundary effects (empirically bounded at $\sim$22\% at $\delta{=}5$\,s, decaying to $<$0.3\% at $\delta{>}60$\,s). This is formalized in \S\ref{sec:deviation}.

\subsection{Pareto Optimality}

A watermark configuration $(C, \tau)$ is \textit{Pareto optimal} if no other configuration simultaneously improves completeness and reduces wait time.
The set of all Pareto-optimal configurations constitutes the \textit{Pareto frontier}.
We examine whether heuristic $+$ DLQ configurations \textbf{Pareto-dominate} strict configurations --- i.e., achieve both higher completeness AND lower wait time.

\textbf{Compression formula} (single divisor, shared reference):
\begin{equation}
  t_{\text{ref}} = \min(\text{pickup\_timestamps}),\quad
  \eventtime = (\text{pickup} - t_{\text{ref}}) / 60,\quad
  \arrivaltime = (\text{dropoff} - t_{\text{ref}}) / 60
\end{equation}
\textbf{Crucial property:} $\lateness = \arrivaltime - \eventtime = (\text{dropoff} - \text{pickup}) / 60$.
Both timestamps share the same $t_{\text{ref}}$ and divisor --- the subtraction cancels the offset entirely.
Lateness = trip duration $\div 60$, preserving the natural out-of-order pattern.

\section{HEURISTIC Watermark: Adaptive Analysis}

\subsection{Algorithm \& DDSketch Mechanism}

\begin{lstlisting}[caption={Heuristic watermark algorithm}, label={lst:heuristic}]
L_eff  = DDSketch.quantile(p_normal)     // streaming p-th percentile
W_h(i) = max_event_time_seen_before_i - L_eff

Events that fail -> DLQ -> eventual reconciliation (C_ev = 100%)
\end{lstlisting}

The DDSketch \cite{masson2019} provides $\varepsilon$-accurate streaming quantile estimates with $O(\log(\max/\min))$ space.
In our simulation, $\Leff$ is recomputed every 50,000 events.

\subsection{Eventual Completeness Guarantee}

\noindent\textbf{Theorem (Informal).}
\textit{With DLQ reconciliation, the heuristic watermark achieves $\Cev = 1.0$ for any $p \in [0, 1]$, independent of the lateness distribution.}

\vspace{4pt}
\noindent\textit{Proof sketch:} Every event is either (1)~correctly classified on first arrival, or (2)~routed to DLQ.
Since DLQ is eventually drained and all stored events are incorporated, $\Cev = \Cimm + (1 - \Cimm) = 1.0$.
The only operational concern is \textit{reconciliation latency} --- a deployment-level SLA parameter, not a completeness concern.

\subsection{Pareto Dominance Analysis}

\textbf{Definition.} Configuration $A$ \textit{Pareto-dominates} $B$ iff:
$C(A) \geq C(B)$, $\tau(A) \leq \tau(B)$, and at least one inequality is strict.

\textbf{Result:} The heuristic $+$ DLQ strategy is \textbf{strictly Pareto-dominant} in the eventual completeness space:
at any $\tau$, $\Cev = 100\%$ for heuristic vs.\ $\leq 99.97\%$ for strict.

When comparing \textbf{immediate completeness only}:
\begin{itemize}[leftmargin=*, itemsep=1pt]
  \item $\tau < 12{,}000$\,ms: Heuristic is Pareto-dominant
  \item $\tau \in [12{,}000, 20{,}000]$\,ms: Neither dominates strictly
  \item $\tau > 20{,}000$\,ms: Heuristic is Pareto-dominant on $\Cev$
\end{itemize}

\subsection{The DLQ as a Game-Changer}

The DLQ fundamentally transforms the trade-off space. Without DLQ:
\begin{equation}
  \max_{\delta} \Cstrict(\delta) \quad \text{s.t.} \quad \tau(\delta) \leq \tau_{\text{SLA}}
\end{equation}
With DLQ, the problem becomes a \textit{two-objective optimization}:
\begin{equation}
  \max_{p} \big(\Cimm(p),\; \text{DLQ\_backlog}(p)\big) \quad \text{s.t.} \quad p \in [0, 1]
\end{equation}
The DLQ backlog $B(p) = N \cdot (1-p)$ decreases with $p$.
For $p{=}0.50$: $B = 1{,}060{,}021$; for $p{=}0.95$: $B = 154{,}984$.

\section{Theoretical Implications \& Discussion}
% ===========================================================================

\subsection{Why Heuristic $+$ DLQ is Theoretically Superior}

The heuristic $+$ DLQ strategy separates two concerns conflated in strict watermarks:
\begin{enumerate}[leftmargin=*, itemsep=2pt]
  \item \textbf{Latency budget} --- controlled by $p_{\text{normal}}$
  \item \textbf{Completeness guarantee} --- handled by DLQ reconciliation
\end{enumerate}
This separation of concerns is a classic systems design principle.
The strict watermark conflates the two: $\delta$ simultaneously determines both latency and completeness.

\subsection{Practical Limitations \& Caveats}

\begin{enumerate}[leftmargin=*, itemsep=2pt]
  \item \textbf{DLQ Reconciliation Latency:} Analysis assumes eventual reconciliation. In practice, reconciliation has its own latency budget between initial emission and final corrected emission.
  \item \textbf{DLQ Storage Cost:} At $p{=}0.50$, 1.06M events in DLQ --- non-trivial for high-throughput deployments.
  \item \textbf{Window Re-emission Semantics:} Corrected windows require downstream consumers to handle updates/retractions.
  \item \textbf{Partition Skew:} 2.21$\times$ partition load skew means some partitions carry disproportionately large DLQ backlogs. Per-partition adaptive $\Leff$ would improve fairness.
  \item \textbf{Time Compression Generality:} DIV$=$60 is dataset-specific. The analysis methodology (Pareto frontier, marginal cost) generalizes to any lateness distribution.
\end{enumerate}

\subsection{Comparison to Prior Art}

Our findings are consistent with the broader stream processing literature:
\begin{itemize}[leftmargin=*, itemsep=2pt]
  \item \textbf{MillWheel} \cite{akidau2013}: Introduced low-watermark concept. Our strict watermark is a direct instantiation; we quantify its completeness-latency trade-off curve.
  \item \textbf{The Dataflow Model} \cite{akidau2015}: Proposed event/processing time separation. Our heuristic model extends this with adaptive $\Leff$.
  \item \textbf{Apache Flink} \cite{carbone2017}: Flink's \texttt{allowedLateness} is analogous to our $\delta$. Our data confirms this single knob is insufficient.
  \item \textbf{DDSketch} \cite{masson2019}: The streaming quantile estimator we simulate. Our convergence analysis adds practical cold-start guidance.
  \item \textbf{Out-of-Order Processing} \cite{li2008}: Characterized the fundamental latency-completeness trade-off.
\end{itemize}

\section{Conclusion}
% ===========================================================================

This report has provided a rigorous empirical and theoretical comparison of strict vs.\ heuristic watermarks for out-of-order stream processing, based on a 2.96M-event dataset with realistic time-compressed lateness patterns.

\subsection*{Key Takeaways}
\begin{enumerate}[leftmargin=*, itemsep=3pt]
  \item \textbf{Strict watermarks exhibit diminishing returns:} Completeness increases from 8.75\% ($\delta{=}0$) to 99.97\% ($\delta{=}120$\,s), but marginal cost grows exponentially: $\sim$200 ms/\% at $\delta{<}10$\,s $\to$ $>$48,000 ms/\% at $\delta{>}60$\,s.
  \item \textbf{Heuristic $+$ DLQ achieves Pareto dominance:} At comparable latency, heuristic always reaches 100\% eventual completeness (vs.\ $\leq$99.97\% for strict). $\Cimm \approx p_{\text{normal}}$.
  \item \textbf{The DLQ separates latency from completeness:} This is the fundamental architectural insight --- strict watermarks conflate the two; heuristic $+$ DLQ decouples them.
  \item \textbf{Balanced recommendation ($p{=}0.90$, $\Leff{\approx}29$\,s):} 91.3\% immediate completeness, 257K-event DLQ backlog (9\%), 100\% eventual accuracy with low overhead.
  \item \textbf{Window size sensitivity is modest:} Completeness curve shape is primarily driven by lateness distribution. Window size can be chosen by application semantics.
\end{enumerate}

\subsection*{Final Recommendation for Topic \#112}

\begin{itemize}[leftmargin=*, itemsep=3pt]
  \item \textbf{Default:} Heuristic $p{=}0.90$ + DLQ --- sweet spot (29\,s latency, 91\% immediate, 100\% eventual)
  \item \textbf{Latency-critical:} Heuristic $p{=}0.50$ --- minimal watermark wait (12\,s), larger DLQ
  \item \textbf{Simplicity-first:} Strict $\delta{=}38$\,s --- predictable, deterministic, 95.7\% completeness
\end{itemize}
